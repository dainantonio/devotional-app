            renderLivingWindow() {
                const svg = document.getElementById('rose-window');
                svg.innerHTML = '';
                const count = app.state.visitCount;
                const center = 50, segments = 30;
                svg.innerHTML += `<circle cx="50" cy="50" r="8" class="window-segment ${count > 0 ? 'lit' : ''}" />`;
                
                for(let i=0; i<segments; i++) {
                    const angle = (i / segments) * Math.PI * 2;
                    const nextAngle = ((i+1) / segments) * Math.PI * 2;
                    const r1 = 12, r2 = 45;
                    const x1 = 50 + r1 * Math.cos(angle), y1 = 50 + r1 * Math.sin(angle);
                    const x2 = 50 + r2 * Math.cos(angle), y2 = 50 + r2 * Math.sin(angle);
                    const x3 = 50 + r2 * Math.cos(nextAngle), y3 = 50 + r2 * Math.sin(nextAngle);
                    const x4 = 50 + r1 * Math.cos(nextAngle), y4 = 50 + r1 * Math.sin(nextAngle);
                    const d = `M ${x1} ${y1} L ${x2} ${y2} A ${r2} ${r2} 0 0 1 ${x3} ${y3} L ${x4} ${y4} A ${r1} ${r1} 0 0 0 ${x1} ${y1}`;
                    const isLit = count > i + 1 ? 'lit' : '';
                    svg.innerHTML += `<path d="${d}" class="window-segment ${isLit}" />`;
                }
            },

            holdScripture(id) {
                const data = DEVOTIONALS.find(d => d.id === id);

                const exists = this.state.heldWords.find(w => w.id === id);

            async generateWord() {
                const input = document.getElementById('generator-input');
                const topic = input.value.trim();
                if(!topic) return;
                
                if(!this.state.apiKey) return ui.showToast('Please add API Key in sidebar settings', 'error');
                
                const btn = document.getElementById('generate-btn');
                const originalText = btn.textContent;
                btn.innerHTML = '<span class="spinner"></span>';
                
                const prompt = `Create a short Christian devotional about "${topic}". Format as JSON: "title", "scripture" (text only), "ref", "reflection" (2 sentences), "prompt", "prayerLines" (array 4 strings). Tone: Calming.`;
                
                try {
                    const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${this.state.apiKey}`, {
                        method: 'POST', headers: {'Content-Type': 'application/json'},
                        body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
                    });
                    const json = await res.json();
                    const raw = json.candidates[0].content.parts[0].text.replace(/```json|```/g, '').trim();
                    const data = JSON.parse(raw);
                    
                    const newId = Date.now();
                    DEVOTIONALS.unshift({ id: newId, tags: ['generated'], ...data, date: "Personal Word" });
                    
                    ui.closeModal('generator');
                    ui.renderContent('generated'); // Force render
                    ui.showToast('Word received', 'success');
                    input.value = '';
                } catch(e) {
                    console.error(e);
                    ui.showToast('Connection error', 'error');
                } finally {
                    btn.textContent = originalText;
                }
            },
            
            toggleAudio() {
                if(!this.audioCtx) this.audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                const btn = document.getElementById('btn-audio');
                
                if(this.state.audioPlaying) {
                    this.audioCtx.suspend();
                    this.state.audioPlaying = false;
                    btn.style.color = 'var(--color-dim)';
                } else {
                    this.audioCtx.resume();
                    if(!this.osc) {
                        // Create Drone
                        [110, 164.8, 196].forEach(f => { // A2, E3, G3
                            const osc = this.audioCtx.createOscillator();
                            const g = this.audioCtx.createGain();
                            osc.type = 'triangle'; osc.frequency.value = f;
                            g.gain.value = 0.03;
                            osc.connect(g); g.connect(this.audioCtx.destination);
                            osc.start();
                        });
                        this.osc = true;
                    }
                    this.state.audioPlaying = true;
                    btn.style.color = 'var(--color-primary)';
                }
            }
