        // --- VISUAL HABIT TRACKER: THE LIVING WINDOW ---
        function renderLivingWindow() {
            const svg = document.getElementById('rose-window');
            svg.innerHTML = ''; // Clear
            
            const daysVisited = state.graceCount; // Using simple count for demo. Ideally unique days.
            const totalSegments = 30; // 30 day cycle
            const center = 50;
            const radius = 45;
            
            // Create Center Rose
            const centerCircle = document.createElementNS("http://www.w3.org/2000/svg", "circle");
            centerCircle.setAttribute("cx", center);
            centerCircle.setAttribute("cy", center);
            centerCircle.setAttribute("r", 10);
            centerCircle.setAttribute("class", "window-segment " + (daysVisited > 0 ? "lit" : ""));
            svg.appendChild(centerCircle);

            // Create Outer Segments
            for(let i=0; i<totalSegments; i++) {
                const angle = (i / totalSegments) * Math.PI * 2;
                const nextAngle = ((i+1) / totalSegments) * Math.PI * 2;
                
                // Draw petal/segment path
                const r1 = 12;
                const r2 = 45;
                
                const x1 = center + r1 * Math.cos(angle);
                const y1 = center + r1 * Math.sin(angle);
                const x2 = center + r2 * Math.cos(angle);
                const y2 = center + r2 * Math.sin(angle);
                const x3 = center + r2 * Math.cos(nextAngle);
                const y3 = center + r2 * Math.sin(nextAngle);
                const x4 = center + r1 * Math.cos(nextAngle);
                const y4 = center + r1 * Math.sin(nextAngle);
                
                const d = `M ${x1} ${y1} L ${x2} ${y2} A ${r2} ${r2} 0 0 1 ${x3} ${y3} L ${x4} ${y4} A ${r1} ${r1} 0 0 0 ${x1} ${y1}`;
                
                const path = document.createElementNS("http://www.w3.org/2000/svg", "path");
                path.setAttribute("d", d);
                path.setAttribute("class", "window-segment " + (daysVisited > i + 1 ? "lit" : ""));
                svg.appendChild(path);
            }
        }

                    textEl.style.opacity = 0;

                        textEl.textContent = phases[0];

                    }, 500);

        function getLiturgicalSeason(now) {
            const year = now.getFullYear();
            const month = now.getMonth(); const day = now.getDate();
            if (month === 0 && day <= 5) return "Christmas Season"; 
            if (month === 0 && day === 6) return "Epiphany";
            if (month === 11 && day >= 25) return "Christmas Season";
            if (month === 11 && day < 25) return "Advent"; 
            
            // Easter Calc
            const a = year % 19, b = Math.floor(year / 100), c = year % 100;
            const d = Math.floor(b / 4), e = b % 4, f = Math.floor((b + 8) / 25);
            const g = Math.floor((b - f + 1) / 3), h = (19 * a + b - d - g + 15) % 30;
            const i = Math.floor(c / 4), k = c % 4, l = (32 + 2 * e + 2 * i - h - k) % 7;
            const m = Math.floor((a + 11 * h + 22 * l) / 451);
            const easterMonth = Math.floor((h + l - 7 * m + 114) / 31) - 1; 
            const easterDay = ((h + l - 7 * m + 114) % 31) + 1;
            const easterDate = new Date(year, easterMonth, easterDay);
            const lentStart = new Date(easterDate); lentStart.setDate(easterDate.getDate() - 46);
            const pentecost = new Date(easterDate); pentecost.setDate(easterDate.getDate() + 49);

            if (now >= lentStart && now < easterDate) return "Lent";
            if (now >= easterDate && now <= pentecost) return "Eastertide";
            return "Ordinary Time";
        }

                    if(isHeld) textEl.classList.add('held');
                }
            }
            typeWord();
        }

        async function callGemini(promptText) {
            if(!apiKey) {
                alert("Please add your Gemini API Key in the Sidebar settings.");
                throw new Error("No API Key");
            }
            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            const payload = { contents: [{ parts: [{ text: promptText }] }] };
            try {
                const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
                if (!response.ok) throw new Error(`API Error: ${response.status}`);
                const data = await response.json();
                return data.candidates[0].content.parts[0].text;
            } catch (error) { console.error("Gemini Error:", error); throw error; }
        }

        async function generateDevotional() {
            const input = document.getElementById('generator-input');

            const btnText = document.getElementById('gen-btn-text');
            const feedback = document.getElementById('gen-feedback');
            btnText.innerHTML = '<span class="spinner"></span> Creating...';
            feedback.style.opacity = '1';
            const prompt = `You are a gentle, wise Christian spiritual director. Create a short devotional about "${topic}". Format as JSON: "title", "scripture" (text only), "ref", "reflection" (2-3 sentences), "prompt", "prayerLines" (array 4 strings), "breathPhrases" (array 2 strings like ["Inhale peace", "Exhale fear"]). Tone: Calming, orthodox.`;
            try {
                const resultText = await callGemini(prompt);
                const cleanJson = resultText.replace(/```json/g, '').replace(/```/g, '').trim();

                const newId = Date.now();

        }

        function generateShareImage(id) {
            const data = DEVOTIONALS.find(d => d.id === id); const canvas = document.getElementById('share-canvas'); const ctx = canvas.getContext('2d'); const overlay = document.getElementById('share-overlay'); const width = 1080; const height = 1350; canvas.width = width; canvas.height = height; const grd = ctx.createLinearGradient(0, 0, width, height); grd.addColorStop(0, data.colors[0]); grd.addColorStop(0.5, data.colors[1]); grd.addColorStop(1, data.colors[2]); ctx.fillStyle = grd; ctx.fillRect(0, 0, width, height); ctx.strokeStyle = "rgba(255,255,255,0.1)"; ctx.lineWidth = 4; ctx.strokeRect(40, 40, width-80, height-80); ctx.fillStyle = '#F7FAFC'; ctx.textAlign = 'center'; ctx.font = '60px Cinzel'; ctx.fillText(data.title.toUpperCase(), width/2, 200); ctx.font = 'italic 50px Cinzel'; wrapText(ctx, '"' + data.scripture + '"', width/2, 400, 800, 80); ctx.fillStyle = '#D4AF37'; ctx.font = '36px Lato'; ctx.fillText(data.ref, width/2, 1000); ctx.fillStyle = '#A0AEC0'; ctx.font = '30px Lato'; ctx.fillText("DAILY GRACE SANCTUARY", width/2, height - 100); overlay.classList.add('active'); const link = document.getElementById('download-link'); link.href = canvas.toDataURL("image/png"); link.download = `daily-grace-${data.date}.png`;

        function wrapText(context, text, x, y, maxWidth, lineHeight) { const words = text.split(' '); let line = ''; for(let n = 0; n < words.length; n++) { const testLine = line + words[n] + ' '; const metrics = context.measureText(testLine); if (metrics.width > maxWidth && n > 0) { context.fillText(line, x, y); line = words[n] + ' '; y += lineHeight; } else { line = testLine; } } context.fillText(line, x, y); return y + lineHeight; }
