        function getLiturgicalSeason(now) {
            const year = now.getFullYear();
            const month = now.getMonth(); 
            const day = now.getDate();
            
            // Fixed dates
            if (month === 0 && day <= 5) return "Christmas Season"; 
            if (month === 0 && day === 6) return "Epiphany";
            if (month === 11 && day >= 25) return "Christmas Season";
            
            // Advent (Approx: 4 Sundays before Christmas)
            if (month === 11 && day < 25) return "Advent"; // Simple proxy for now

            // Easter Calculation (Meeus/Jones/Butcher)
            const a = year % 19;
            const b = Math.floor(year / 100);
            const c = year % 100;
            const d = Math.floor(b / 4);
            const e = b % 4;
            const f = Math.floor((b + 8) / 25);
            const g = Math.floor((b - f + 1) / 3);
            const h = (19 * a + b - d - g + 15) % 30;
            const i = Math.floor(c / 4);
            const k = c % 4;
            const l = (32 + 2 * e + 2 * i - h - k) % 7;
            const m = Math.floor((a + 11 * h + 22 * l) / 451);
            const easterMonth = Math.floor((h + l - 7 * m + 114) / 31) - 1; 
            const easterDay = ((h + l - 7 * m + 114) % 31) + 1;
            
            const easterDate = new Date(year, easterMonth, easterDay);
            
            // Lent: 46 days before Easter
            const lentStart = new Date(easterDate);
            lentStart.setDate(easterDate.getDate() - 46);
            
            // Pentecost: 50 days after Easter
            const pentecost = new Date(easterDate);
            pentecost.setDate(easterDate.getDate() + 49);

            if (now >= lentStart && now < easterDate) return "Lent";
            if (now >= easterDate && now <= pentecost) return "Eastertide";
            
            return "Ordinary Time";
        }

        function startLectioMode(id) {
            const data = DEVOTIONALS.find(d => d.id === id);
            const container = document.getElementById(`scripture-box-${id}`);

            let i = 0;
            
            function typeWord() {
                if(i < words.length) {
                    const span = document.createElement("span");
                    span.textContent = words[i] + " ";
                    span.style.opacity = "0";
                    span.style.transition = "opacity 0.5s ease";
                    textEl.appendChild(span);
                    
                    // Trigger fade in
                    requestAnimationFrame(() => span.style.opacity = "1");
                    
                    i++;
                    setTimeout(typeWord, 600); // Slow pace: 600ms per word
                } else {
                    // Restore hold status if applicable
                    const isHeld = state.heldWords.find(w => w.id === id);
                    if(isHeld) textEl.classList.add('held');
                }
            }
            typeWord();
        }

        async function callGemini(promptText) {
            if(!apiKey) {
                alert("Please add your Gemini API Key in the Sidebar settings.");
                throw new Error("No API Key");
            }
            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            const payload = {
                contents: [{ parts: [{ text: promptText }] }]
            };
            
            try {
                const response = await fetch(url, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                
                if (!response.ok) throw new Error(`API Error: ${response.status}`);
                const data = await response.json();
                return data.candidates[0].content.parts[0].text;
            } catch (error) {
                console.error("Gemini Error:", error);
                throw error;
            }
        }

        function wrapText(context, text, x, y, maxWidth, lineHeight) {
            const words = text.split(' ');
            let line = '';
            for(let n = 0; n < words.length; n++) {
                const testLine = line + words[n] + ' ';
                const metrics = context.measureText(testLine);
                if (metrics.width > maxWidth && n > 0) {
                    context.fillText(line, x, y);
                    line = words[n] + ' ';
                    y += lineHeight;
                } else { line = testLine; }
            }
            context.fillText(line, x, y);
            return y + lineHeight; 
        }

        // --- DYNAMIC ATMOSPHERE ---
        function setupObserver() {
            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        entry.target.classList.add('active');
                        const colors = JSON.parse(entry.target.dataset.colors);
                        document.documentElement.style.setProperty('--atm-1', colors[0]);
                        document.documentElement.style.setProperty('--atm-2', colors[1]);
                        document.documentElement.style.setProperty('--atm-3', colors[2]);
                    }
                });
            }, { threshold: 0.5 });
            document.querySelectorAll('.slide').forEach(slide => observer.observe(slide));
        }

                textEl.classList.remove('held');
            }
            localStorage.setItem('dailyGrace_held', JSON.stringify(state.heldWords));
            renderHeldWords();
        }

        function renderHeldWords() {
            const list = document.getElementById('held-list');
            if (state.heldWords.length === 0) {
                list.innerHTML = '<p style="font-size: 0.8rem; color: #555;">Double-tap verse text to hold...</p>';
                return;
            }
            list.innerHTML = '';
            state.heldWords.forEach(item => {
                const div = document.createElement('div');
                div.innerHTML = `<p style="font-size:0.9rem; color:#ddd; font-style:italic; margin-bottom:5px;">"${item.text.substring(0, 50)}..."</p><span style="font-size:0.75rem; color:var(--text-dim);">${item.ref}</span>`;
                div.style.marginBottom = '20px';
                list.appendChild(div);
            });
        }
        
        function toggleBreath() {
            state.breathEnabled = !state.breathEnabled;
            document.getElementById('breath-toggle').classList.toggle('active');
            document.querySelectorAll('.scripture-container').forEach(el => {
                if (state.breathEnabled) el.classList.add('breathe-active');
                else el.classList.remove('breathe-active');
            });
        }

        function toggleSidebar() {
            document.getElementById('sidebar').classList.toggle('open');
            document.getElementById('overlay').classList.toggle('active');
        }

        function createParticles() {
            const bg = document.getElementById('bg-layer');
            for(let i=0; i<20; i++) {
                const p = document.createElement('div');
                p.className = 'particle';
                const size = Math.random() * 4 + 1;
                p.style.width = `${size}px`;
                p.style.height = `${size}px`;
                p.style.left = `${Math.random() * 100}%`;
                p.style.animationDuration = `${Math.random() * 15 + 10}s`;
                p.style.animationDelay = `${Math.random() * 5}s`;
                bg.appendChild(p);
            }
        }

        function toggleAudio() {
            const btn = document.getElementById('audio-btn');
            if (state.isAudioPlaying) {
                state.audioContext.close();
                state.isAudioPlaying = false;
                btn.classList.remove('active');
            } else {
                state.audioContext = new (window.AudioContext || window.webkitAudioContext)();
                const freq = [164.81, 207.65, 246.94]; 
                freq.forEach(f => {
                    const osc = state.audioContext.createOscillator();
                    const gain = state.audioContext.createGain();
                    osc.type = 'triangle'; osc.frequency.value = f;
                    const lfo = state.audioContext.createOscillator();
                    lfo.type = 'sine'; lfo.frequency.value = 0.1 + Math.random() * 0.1;
                    const lfoGain = state.audioContext.createGain();
                    lfoGain.gain.value = 0.02;
                    osc.connect(gain); lfo.connect(lfoGain); lfoGain.connect(gain.gain);
                    gain.connect(state.audioContext.destination); gain.gain.value = 0.03; 
                    osc.start(); lfo.start();
                });
                state.isAudioPlaying = true;
                btn.classList.add('active');
            }
        }
