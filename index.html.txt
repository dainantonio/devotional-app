            const ripple = document.createElement('div');
            ripple.className = 'ripple';
            btn.parentElement.appendChild(ripple);

        // --- DYNAMIC ATMOSPHERE OBSERVER ---
        function setupObserver() {
            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        entry.target.classList.add('active');
                        
                        // Update CSS Variables for Background Gradient
                        const colors = JSON.parse(entry.target.dataset.colors);
                        document.documentElement.style.setProperty('--atm-1', colors[0]);
                        document.documentElement.style.setProperty('--atm-2', colors[1]);
                        document.documentElement.style.setProperty('--atm-3', colors[2]);
                    }
                });
            }, { threshold: 0.5 });
            
            document.querySelectorAll('.slide').forEach(slide => observer.observe(slide));
        }

        // --- SILENCE TIMER ---
        function startSilence() {
            const overlay = document.getElementById('silence-overlay');
            const display = document.getElementById('timer-display');
            overlay.classList.add('active');
            let timeLeft = 60;
            display.textContent = "01:00";
            if(state.silenceInterval) clearInterval(state.silenceInterval);
            
            state.silenceInterval = setInterval(() => {
                timeLeft--;
                const m = Math.floor(timeLeft / 60);
                const s = timeLeft % 60;
                display.textContent = `${m < 10 ? '0' : ''}${m}:${s < 10 ? '0' : ''}${s}`;
                if (timeLeft <= 0) {
                    clearInterval(state.silenceInterval);
                    display.textContent = "Amen.";
                }
            }, 1000);
        }

        function stopSilence() {
            document.getElementById('silence-overlay').classList.remove('active');
            if(state.silenceInterval) clearInterval(state.silenceInterval);
        }

        function toggleHold(id, container) {
            const textEl = container.querySelector('.scripture-text');
            const data = DEVOTIONALS.find(d => d.id === id);
            const index = state.heldWords.findIndex(w => w.id === id);

            if (index === -1) {
                state.heldWords.push({ id: data.id, text: data.scripture, ref: data.ref });
                textEl.classList.add('held');
            } else {
                state.heldWords.splice(index, 1);
                textEl.classList.remove('held');
            }

            localStorage.setItem('dailyGrace_held', JSON.stringify(state.heldWords));
            renderHeldWords();
        }

        function renderHeldWords() {
            const list = document.getElementById('held-list');
            if (state.heldWords.length === 0) {
                list.innerHTML = '<p style="font-size: 0.8rem; color: #555;">Double-tap verse text to hold...</p>';
                return;
            }
            list.innerHTML = '';
            state.heldWords.forEach(item => {
                const div = document.createElement('div');
                div.innerHTML = `<p style="font-size:0.9rem; color:#ddd; font-style:italic; margin-bottom:5px;">"${item.text.substring(0, 50)}..."</p><span style="font-size:0.75rem; color:var(--text-dim);">${item.ref}</span>`;
                div.style.marginBottom = '20px';
                list.appendChild(div);
            });
        }

        function toggleAudio() {
            const btn = document.getElementById('audio-btn');
            if (state.isAudioPlaying) {
                state.audioContext.close();
                state.isAudioPlaying = false;
                btn.classList.remove('active');
            } else {
                state.audioContext = new (window.AudioContext || window.webkitAudioContext)();
                const freq = [164.81, 207.65, 246.94]; 
                freq.forEach(f => {
                    const osc = state.audioContext.createOscillator();
                    const gain = state.audioContext.createGain();
                    osc.type = 'triangle'; osc.frequency.value = f;
                    const lfo = state.audioContext.createOscillator();
                    lfo.type = 'sine'; lfo.frequency.value = 0.1 + Math.random() * 0.1;
                    const lfoGain = state.audioContext.createGain();
                    lfoGain.gain.value = 0.02;
                    osc.connect(gain); lfo.connect(lfoGain); lfoGain.connect(gain.gain);
                    gain.connect(state.audioContext.destination); gain.gain.value = 0.03; 
                    osc.start(); lfo.start();
                });
                state.isAudioPlaying = true;
                btn.classList.add('active');
            }
        }
